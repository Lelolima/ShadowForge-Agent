// BOOT SEQUENCE — QUICK START

# 1. Clone & configure
$ git clone https://github.com/shadowforge/NVIDIA-ShadowForge-Agent
$ cd ShadowForge-Agent
$ cp .env.example .env
# 2. Set NVIDIA_API_KEY in .env
# 3. Install dependencies
$ pip install -r requirements.txt
$ pip install -r requirements-hacker.txt # pentest tools
$ pip install -r requirements-vision.txt # vision stack
# 4. Launch agent
$ python main.py --mode stealth --target 192.168.1.0/24
$ python main.py --voice --always-listen # voice mode
$ python main.py --simulate # safe simulation
